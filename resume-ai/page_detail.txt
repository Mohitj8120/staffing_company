Template-based: Ek specially prepared template jisme placeholders hon. Isme formatting preserve hogi, lekin template ko ek baar manually prepare karna padega.

Ye decision Phase 4 ke code ko completely affect karta hai. Agar bina decide kiye main code de dunga to baad mein poora generator rewrite karna padega.

Option B: Existing resume DOCX ko directly modify karenge (complex, but no placeholder setup).

import os

from docxtpl import DocxTemplate

from config import (
    TEMPLATE_FOLDER,
    OUTPUT_FOLDER
)


def create_resume(data):

    template = os.path.join(
        TEMPLATE_FOLDER,
        "resume_template.docx"
    )

    doc = DocxTemplate(template)

    context = {

        "name":
        data["personal"]["name"],

        "email":
        data["personal"]["email"],

        "phone":
        data["personal"]["phone"],

        "summary":
        data["summary"],

        "skills":
        "\n".join(data["skills"]),

        "experience":
        "\n\n".join(
            [
                f'{e["title"]}\n{e["company"]}\n{e["duration"]}\n' +
                "\n".join(
                    ["• "+p for p in e["points"]]
                )
                for e in data["experience"]
            ]
        ),

        "projects":
        "\n\n".join(
            [
                f'{p["title"]}\n{p["description"]}'
                for p in data["projects"]
            ]
        ),

        "education":
        "\n\n".join(
            [
                f'{e["degree"]}\n{e["college"]}\n{e["year"]}'
                for e in data["education"]
            ]
        ),

        "certifications":
        "\n".join(
            data["certifications"]
        )

    }

    doc.render(context)

    output = os.path.join(
        OUTPUT_FOLDER,
        "tailored_resume.docx"
    )

    doc.save(output)

    return output

from fastapi import APIRouter
from fastapi import Form

import os

from config import OUTPUT_FOLDER

from utils.json_utils import (
    load_json,
    save_json
)

from services.resume_optimizer import optimize_resume
from services.resume_generator import create_resume
from utils.pdf_converter import create_pdf

router = APIRouter()


@router.post("/optimize")

def optimize(

        jd: str = Form(...)

):

    resume = load_json(

        os.path.join(

            OUTPUT_FOLDER,

            "resume.json"

        )

    )

    optimized = optimize_resume(

        resume,

        jd

    )

    save_json(

        optimized,

        os.path.join(

            OUTPUT_FOLDER,

            "optimized_resume.json"

        )

    )

    docx = create_resume(

        optimized

    )

    pdf = os.path.join(

        OUTPUT_FOLDER,

        "tailored_resume.pdf"

    )

    create_pdf(

        docx,

        pdf

    )

    return {

        "status":"success",

        "pdf":"Resume Generated"

    }
